model Order {

  paidAt            DateTime?
  
  // Wysyłka
  shippingAddressId String
  billingAddressId  String

  deliveredAt       DateTime?

  dueDate           DateTime

  actualEnd        DateTime?

  issues           String?          @db.Text

  // Relacje
