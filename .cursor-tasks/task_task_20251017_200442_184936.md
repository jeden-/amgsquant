```typescript
import { PrismaClient } from '@prisma/client';
import * as bcrypt from 'bcrypt';
import {
  materials,
  products,
  pricingRulesPosters,
  pricingRulesFoils,
  users,
  machines,
} from '../src/seed-data';

const prisma = new PrismaClient();

async function main() {
  console.log('🌱 Starting database seed...\n');

  // 1. Czyszczenie bazy (opcjonalne - użyj ostrożnie!)
  console.log('🗑️  Clearing existing data...');
  await prisma.orderStatusHistory.deleteMany();
  await prisma.orderItem.deleteMany();
  await prisma.productionJob.deleteMany();
  await prisma.order.deleteMany();
  await prisma.stockMovement.deleteMany();
  await prisma.file.deleteMany();
  await prisma.pricingRule.deleteMany();
  await prisma.machine.deleteMany();
  await prisma.material.deleteMany();
  await prisma.product.deleteMany();
  await prisma.address.deleteMany();
  await prisma.user.deleteMany();
  await prisma.dailyStats.deleteMany();
  console.log('✅ Data cleared\n');

  // 2. Materiały
  console.log('📦 Seeding materials...');
  for (const material of materials) {
    await prisma.material.create({ data: material });
  }
  console.log(`✅ Created ${materials.length} materials\n`);

  // 3. Produkty
  console.log('🛍️  Seeding products...');
  const createdProducts = [];
  for (const product of products) {
    const created = await prisma.product.create({ data: product });
    createdProducts.push(created);
  }
  console.log(`✅ Created ${products.length} products\n`);

  // 4. Cenniki (PricingRules) - powiązane z produktami
  console.log('💰 Seeding pricing rules...');
  const posterProduct = createdProducts.find(p => p.type === 'POSTER');
  const stickerProduct = createdProducts.find(p => p.type === 'STICKER');

  if (posterProduct) {
    for (const rule of pricingRulesPosters) {
      await prisma.pricingRule.create({
        data: {
          ...rule,
          product: { connect: { id: posterProduct.id } },
        },
      });
    }
  }

  if (stickerProduct) {
    for (const rule of pricingRulesFoils) {
      await prisma.pricingRule.create({
        data: {
          ...rule,
          product: { connect: { id: stickerProduct.id } },
        },
      });
    }
  }
  console.log(`✅ Created pricing rules\n`);

  // 5. Użytkownicy
  console.log('👥 Seeding users...');
  const password = 'Test1234!';
  const hashedPassword = await bcrypt.hash(password, 10);

  for (const user of users) {
    const createdUser = await prisma.user.create({
      data: {
        ...user,
        passwordHash: hashedPassword,
      },
    });

    // Dodaj przykładowy adres dla każdego użytkownika
    await prisma.address.create({
      data: {
        userId: createdUser.id,
        type: 'BOTH',
        firstName: user.firstName,
        lastName: user.lastName,
        street: 'ul. Testowa',
        buildingNo: '123',
        apartmentNo: '45',
        city: 'Kraków',
        postalCode: '30-001',
        country: 'PL',
        phone: user.phone || '+48123456789',
        isDefault: true,
        companyName: user.companyName,
      },
    });
  }
  console.log(`✅ Created ${users.length} users (password: ${password})\n`);

  // 6. Maszyny
  console.log('🖨️  Seeding machines...');
  for (const machine of machines) {
    await prisma.machine.create({ data: machine });
  }
  console.log(`✅ Created ${machines.length} machines\n`);

  // 7. Statystyki - utwórz pusty wpis na dzisiaj
  console.log('📊 Creating daily stats entry...');
  await prisma.dailyStats.create({
    data: {
      date: new Date(),
      ordersCount: 0,
      revenue: 0,
      m2Printed: 0,
      jobsCompleted: 0,
      materialUsed: 0,
    },
  });
  console.log('✅ Daily stats created\n');

  console.log('🎉 Seed completed successfully!\n');
  console.log('📝 Login credentials:');
  console.log('   Admin: admin@amgsquant.pl / Test1234!');
  console.log('   B2B: jan.kowalski@firma.pl / Test1234!');
  console.log('   Agency: agencja@reklama.pl / Test1234!');
  console.log('   Customer: klient@example.com / Test1234!\n');
}

main()
  .catch((e) => {
    console.error('❌ Error seeding database:', e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });
```

# @amgsquant/database

Database package zawierający Prisma schema, migracje i seed data dla systemu drukarni.

## Instalacja

```bash
cd packages/database
pnpm install
```

## Dostępne komendy

```bash
# Wygeneruj Prisma Client
pnpm db:generate

# Push schema do bazy (dev)
pnpm db:push

# Stwórz migrację
pnpm db:migrate

# Deploy migracji (produkcja)
pnpm db:migrate:deploy

# Wypełnij bazę przykładowymi danymi
pnpm db:seed

# Otwórz Prisma Studio
pnpm db:studio

# Resetuj bazę (usuń wszystko i zaaplikuj migracje)
pnpm db:reset
```

## Schema

- **User** - Użytkownicy (B2C, B2B, Agency)
- **Address** - Adresy wysyłki/rozliczeniowe
- **Product** - Produkty (Plakaty, Banery, Roll-upy, Naklejki)
- **Material** - Materiały drukarskie (papiery, folie, banery)
- **PricingRule** - Reguły cenowe
- **Order** - Zamówienia
- **OrderItem** - Pozycje zamówień
- **File** - Pliki graficzne
- **ProductionJob** - Zadania produkcyjne
- **Machine** - Maszyny (plottery, laminator, cutter)
- **StockMovement** - Ruchy magazynowe
- **DailyStats** - Statystyki dzienne
