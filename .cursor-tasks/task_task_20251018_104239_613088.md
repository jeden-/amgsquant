  private calculatePrintCost(totalAreaM2: number): number {
    // Koszt druku ~10 PLN/m²
    return totalAreaM2 * 10;
  }

  private calculateFinishingCost(
    options: any,
    areaM2: number,
    quantity: number,
    width: number,
    height: number
  ): number {
    let cost = 0;

    // Laminat
    if (options.laminate) {
      const laminateCost = LAMINATE_COST[options.laminate] || 0;
      cost += laminateCost * areaM2 * quantity;
    }

    // Oczka
    if (options.eyelets) {
      cost += options.eyelets * EYELET_COST * quantity;
    }

    // Tunel
    if (options.tunnel) {
      const widthM = width / 1000;
      cost += widthM * TUNNEL_COST * quantity;
    }

    // Mounting (wyklejanie na podkład)
    if (options.mounting) {
      const mountingCosts = {
        PCV3MM: 65,
        PCV5MM: 75,
        FOAM5MM: 40,
        HIPS3MM: 110,
        DIBOND: 140,
      };
      const mountingCost = mountingCosts[options.mounting] || 0;
      cost += mountingCost * areaM2 * quantity;
    }

    return cost;
  }

  private calculateProductionTime(
    totalAreaM2: number,
    options: any,
    rushType: RushType
  ): number {
    // Czas druku: ~12 m²/h dla plotera
    const printTimeMin = (totalAreaM2 / 12) * 60;

    // Czas wykończenia
    let finishingTimeMin = 0;
    if (options.laminate) finishingTimeMin += totalAreaM2 * 5; // 5 min/m²
    if (options.eyelets) finishingTimeMin += options.eyelets * 2; // 2 min/oczko
    if (options.mounting) finishingTimeMin += totalAreaM2 * 15; // 15 min/m²

    const totalTime = printTimeMin + finishingTimeMin;

    // Bufor 15%
    return Math.ceil(totalTime * 1.15);
  }

  private calculateDeliveryDate(rushType: RushType, productionTimeMinutes: number): Date {
    const now = new Date();
    
    // Konwersja czasu produkcji na godziny
    const productionHours = Math.ceil(productionTimeMinutes / 60);
    
    // Dodaj czas zgodnie z typem rush
    let deliveryHours = 0;
    switch (rushType) {
      case RushType.SAME_DAY:
        deliveryHours = 8;
        break;
      case RushType.EXPRESS_24H:
        deliveryHours = 24;
        break;
      case RushType.EXPRESS_48H:
        deliveryHours = 48;
        break;
      case RushType.STANDARD:
      default:
        deliveryHours = 72;
        break;
    }

    // Dodaj czas produkcji do czasu dostawy
    const totalHours = Math.max(deliveryHours, productionHours + 4); // minimum 4h bufor

    const deliveryDate = new Date(now.getTime() + totalHours * 60 * 60 * 1000);
    return deliveryDate;
  }
}

import { Controller, Post, Get, Body } from '@nestjs/common';
import { ApiTags, ApiOperation, ApiResponse } from '@nestjs/swagger';
import { PricingService } from './pricing.service';
import { CalculatePriceDto } from './dto/calculate-price.dto';
import { PriceResponseDto } from './dto/price-response.dto';

@ApiTags('pricing')
@Controller('pricing')
export class PricingController {
  constructor(private pricingService: PricingService) {}

  @Post('calculate')
  @ApiOperation({ summary: 'Calculate price for product configuration' })
  @ApiResponse({ status: 200, description: 'Price calculated successfully', type: PriceResponseDto })
  async calculatePrice(@Body() dto: CalculatePriceDto): Promise<PriceResponseDto> {
    return await this.pricingService.calculatePrice(dto);
  }

  @Get('materials')
  @ApiOperation({ summary: 'Get available materials with base prices' })
  async getMaterials() {
    return await this.pricingService.getMaterials();
  }

  @Get('rush-options')
  @ApiOperation({ summary: 'Get rush delivery options' })
  async getRushOptions() {
    return await this.pricingService.getRushOptions();
  }

  @Get('finishing-options')
  @ApiOperation({ summary: 'Get finishing options with costs' })
  async getFinishingOptions() {
    return await this.pricingService.getFinishingOptions();
  }
}
